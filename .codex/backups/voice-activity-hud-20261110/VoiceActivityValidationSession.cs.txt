using System;
using System.IO;
using System.Linq;
using System.Reflection;
using NUnit.Framework;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.SceneManagement;
using UnityEngine.UI;
using Object = UnityEngine.Object;

[InitializeOnLoad]
internal static class VoiceActivityValidationSession
{
    private const string Request = "Temp/VoiceActivityValidation.request";
    private const string Results = ".codex/diagnostics/voice-activity-hud-20261110.txt";
    private const string Prefab = "Assets/Resources/UI/VoiceActivityHUD.prefab";

    static VoiceActivityValidationSession() { EditorApplication.update += Update; }

    private static void Update()
    {
        if (!File.Exists(Request) || EditorApplication.isCompiling || EditorApplication.isUpdating
            || EditorApplication.isPlayingOrWillChangePlaymode) return;
        string command = File.ReadAllText(Request).Trim();
        File.Delete(Request);
        Scene original = SceneManager.GetActiveScene();
        bool wasDirty = original.isDirty;
        Scene validation = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Additive);
        SceneManager.SetActiveScene(validation);
        try
        {
            Log("ORIGINAL " + original.path + " dirty=" + wasDirty);
            if (command == "setup-tests") BuildPrefab();
            Tests(typeof(VoiceActivityIndicatorTests));
            Tests(typeof(VivoxVoiceOutputTests));
            Preview();
        }
        catch (Exception error) { Log("VALIDATION FAILED " + error); }
        finally
        {
            SceneManager.SetActiveScene(original);
            EditorSceneManager.CloseScene(validation, true);
            if (wasDirty && !original.isDirty) EditorSceneManager.MarkSceneDirty(original);
            Log("RESTORED " + original.path + " dirty=" + original.isDirty);
        }
    }

    private static void BuildPrefab()
    {
        if (File.Exists(Prefab)) { Log("PREFAB already exists; retaining authored values"); return; }
        var root = new GameObject("Voice Activity HUD", typeof(RectTransform), typeof(Canvas),
            typeof(CanvasScaler), typeof(CanvasGroup));
        root.SetActive(false);
        root.layer = 5;
        Canvas canvas = root.GetComponent<Canvas>();
        canvas.renderMode = RenderMode.ScreenSpaceOverlay;
        canvas.sortingOrder = 50;
        CanvasScaler scaler = root.GetComponent<CanvasScaler>();
        scaler.uiScaleMode = CanvasScaler.ScaleMode.ScaleWithScreenSize;
        scaler.referenceResolution = new Vector2(1920f, 1080f);
        scaler.matchWidthOrHeight = 0.5f;
        root.GetComponent<CanvasGroup>().interactable = false;
        root.GetComponent<CanvasGroup>().blocksRaycasts = false;
        var microphone = new GameObject("Microphone", typeof(RectTransform), typeof(CanvasRenderer),
            typeof(MicrophoneActivityGraphic), typeof(Shadow));
        microphone.layer = 5;
        microphone.transform.SetParent(root.transform, false);
        var rect = (RectTransform)microphone.transform;
        rect.anchorMin = rect.anchorMax = Vector2.zero;
        rect.anchoredPosition = new Vector2(48f, 48f);
        rect.sizeDelta = new Vector2(36f, 40f);
        var graphic = microphone.GetComponent<MicrophoneActivityGraphic>();
        graphic.raycastTarget = false;
        graphic.color = new Color(0.65f, 0.68f, 0.72f, 0.55f);
        graphic.CrossedOut = true;
        Shadow shadow = microphone.GetComponent<Shadow>();
        shadow.effectDistance = new Vector2(1.5f, -1.5f);
        shadow.effectColor = new Color(0f, 0f, 0f, 0.85f);
        var indicator = root.AddComponent<VoiceActivityIndicator>();
        var serialized = new SerializedObject(indicator);
        serialized.FindProperty("microphoneIcon").objectReferenceValue = graphic;
        serialized.ApplyModifiedPropertiesWithoutUndo();
        root.SetActive(true);
        PrefabUtility.SaveAsPrefabAsset(root, Prefab);
        Object.DestroyImmediate(root);
        Log("CREATED " + Prefab);
    }

    private static void Tests(Type type)
    {
        int passed = 0, failed = 0;
        foreach (MethodInfo method in type.GetMethods())
        {
            var cases = method.GetCustomAttributes(typeof(TestCaseAttribute), false).Cast<TestCaseAttribute>().ToArray();
            if (cases.Length == 0 && method.GetCustomAttributes(typeof(TestAttribute), false).Length == 0) continue;
            var arguments = cases.Length > 0 ? cases.Select(c => c.Arguments) : new[] { Array.Empty<object>() };
            foreach (object[] args in arguments)
            {
                object fixture = Activator.CreateInstance(type);
                string name = type.Name + "." + method.Name + "(" + string.Join(",", args) + ")";
                try
                {
                    type.GetMethod("SetUp")?.Invoke(fixture, null);
                    method.Invoke(fixture, args);
                    passed++; Log("PASS " + name);
                }
                catch (Exception error) { failed++; Log("FAIL " + name + " " + (error.InnerException ?? error)); }
                finally { type.GetMethod("TearDown")?.Invoke(fixture, null); }
            }
        }
        Log("TESTS " + type.Name + " passed=" + passed + " failed=" + failed);
    }

    private static void Preview()
    {
        var cameraRoot = new GameObject("Isolated voice HUD preview camera");
        Camera camera = cameraRoot.AddComponent<Camera>();
        camera.clearFlags = CameraClearFlags.SolidColor;
        camera.backgroundColor = new Color(0.055f, 0.065f, 0.08f, 1f);
        camera.cullingMask = 1 << 31;
        camera.orthographic = true;
        camera.nearClipPlane = 0.01f;
        var target = new RenderTexture(1280, 320, 24);
        camera.targetTexture = target;
        string[] labels = { "IDLE", "SPEAKING", "MUTED", "OFFLINE" };
        var snapshots = new[]
        {
            VoiceActivitySnapshot.FromInput(true, true, true, false, true, false, 0d),
            VoiceActivitySnapshot.FromInput(true, true, true, false, true, true, 0.6d),
            VoiceActivitySnapshot.FromInput(true, true, true, true, true, true, 0.6d),
            default(VoiceActivitySnapshot)
        };
        for (int i = 0; i < labels.Length; i++)
        {
            var root = Object.Instantiate(Resources.Load<GameObject>("UI/VoiceActivityHUD"));
            foreach (Transform child in root.GetComponentsInChildren<Transform>(true)) child.gameObject.layer = 31;
            Canvas canvas = root.GetComponent<Canvas>();
            canvas.renderMode = RenderMode.ScreenSpaceCamera;
            canvas.worldCamera = camera;
            canvas.planeDistance = 1f;
            root.GetComponent<CanvasScaler>().uiScaleMode = CanvasScaler.ScaleMode.ConstantPixelSize;
            var icon = root.GetComponentInChildren<MicrophoneActivityGraphic>();
            var rect = (RectTransform)icon.transform;
            rect.anchorMin = rect.anchorMax = new Vector2((i + 0.5f) / 4f, 0.57f);
            rect.anchoredPosition = Vector2.zero;
            rect.sizeDelta = new Vector2(96f, 108f);
            var indicator = root.GetComponent<VoiceActivityIndicator>();
            typeof(VoiceActivityIndicator).GetMethod("Present", BindingFlags.Instance | BindingFlags.NonPublic)
                .Invoke(indicator, new object[] { snapshots[i], 2f });
            var labelRoot = new GameObject("Preview caption", typeof(RectTransform), typeof(Text));
            labelRoot.layer = 31;
            labelRoot.transform.SetParent(root.transform, false);
            var label = labelRoot.GetComponent<Text>();
            label.font = Resources.GetBuiltinResource<Font>("LegacyRuntime.ttf");
            label.text = labels[i]; label.fontSize = 18;
            label.color = new Color(0.7f, 0.75f, 0.8f, 1f);
            label.alignment = TextAnchor.MiddleCenter;
            label.raycastTarget = false;
            var labelRect = (RectTransform)labelRoot.transform;
            labelRect.anchorMin = labelRect.anchorMax = new Vector2((i + 0.5f) / 4f, 0.2f);
            labelRect.anchoredPosition = Vector2.zero;
            labelRect.sizeDelta = new Vector2(240f, 40f);
        }
        Canvas.ForceUpdateCanvases();
        camera.Render();
        RenderTexture previous = RenderTexture.active;
        RenderTexture.active = target;
        var image = new Texture2D(1280, 320, TextureFormat.RGB24, false);
        image.ReadPixels(new Rect(0, 0, 1280, 320), 0, 0); image.Apply();
        File.WriteAllBytes(".codex/diagnostics/voice-activity-hud-preview-20261110.png", image.EncodeToPNG());
        RenderTexture.active = previous;
        camera.targetTexture = null;
        Object.DestroyImmediate(image); Object.DestroyImmediate(target);
        Log("PREVIEW captured with synthetic states, no voice connection or microphone capture");
    }

    private static void Log(string value)
    {
        Directory.CreateDirectory(".codex/diagnostics");
        File.AppendAllText(Results, value + Environment.NewLine);
    }
}
