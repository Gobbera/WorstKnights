    private static void Update()
    {
        if (!File.Exists(Request) || EditorApplication.isCompiling || EditorApplication.isUpdating
            || EditorApplication.isPlayingOrWillChangePlaymode) return;
        string command = File.ReadAllText(Request).Trim(); File.Delete(Request);
        Scene original = SceneManager.GetActiveScene(); bool dirty = original.isDirty;
        Scene validation = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Additive);
        SceneManager.SetActiveScene(validation);
        try
        {
            Log("ORIGINAL " + original.path + " dirty=" + dirty);
            if (command == "setup-tests") BuildPrefab();
            if (command == "fix-style") FixStyle();
            Tests(typeof(VoiceParticipantControlsTests));
            Tests(typeof(VivoxMicrophoneLifecycleTests));
            Tests(typeof(VoiceTransmissionTests));
            Tests(typeof(VoiceActivityIndicatorTests));
            Tests(typeof(VivoxVoiceOutputTests));
            Preview();
            Log("SERVICES=" + Unity.Services.Core.UnityServices.State);
        }
        catch (Exception error) { Log("VALIDATION FAILED " + error); }
        finally
        {
            SceneManager.SetActiveScene(original); EditorSceneManager.CloseScene(validation, true);
            if (dirty && !original.isDirty) EditorSceneManager.MarkSceneDirty(original);
            Log("RESTORED " + original.path + " dirty=" + original.isDirty);
        }
    }

    private static void FixStyle()
    {
        var root = PrefabUtility.LoadPrefabContents(Prefab);
        try
        {
            var slider = root.GetComponentInChildren<Slider>(true);
            if (slider.fillRect.parent == slider.transform)
            {
                var area = Rect("Fill Area", slider.transform, new Vector2(0.5f, 0.5f), Vector2.zero, Vector2.zero);
                area.anchorMin = new Vector2(0f, 0.42f); area.anchorMax = new Vector2(1f, 0.58f);
                slider.fillRect.SetParent(area, false); Stretch(slider.fillRect, Vector2.zero, Vector2.zero);
            }
            slider.handleRect.sizeDelta = new Vector2(14f, 0f);
            slider.SetValueWithoutNotify(1f);
            PrefabUtility.SaveAsPrefabAsset(root, Prefab);
        }
        finally { PrefabUtility.UnloadPrefabContents(root); }
    }

    private static RectTransform Rect(string name, Transform parent, Vector2 anchor, Vector2 size, Vector2 position)
    {
        var root = new GameObject(name, typeof(RectTransform)); root.layer = 5;
        var rect = (RectTransform)root.transform; if (parent != null) rect.SetParent(parent, false);
        rect.anchorMin = rect.anchorMax = anchor; rect.sizeDelta = size; rect.anchoredPosition = position;
        return rect;
    }

    private static TMP_Text Label(string text, Transform parent, Vector2 anchor, Vector2 size, Vector2 position, float fontSize)
    {
        var rect = Rect(text, parent, anchor, size, position);
        var label = rect.gameObject.AddComponent<TextMeshProUGUI>();
        label.font = font; label.text = text; label.fontSize = fontSize; label.color = Color.white;
        label.textWrappingMode = TextWrappingModes.NoWrap; label.raycastTarget = false;
        label.alignment = TextAlignmentOptions.MidlineLeft;
        return label;
    }

    private static void Stretch(RectTransform rect, Vector2 minimum, Vector2 maximum)
    {
        rect.anchorMin = Vector2.zero; rect.anchorMax = Vector2.one; rect.offsetMin = minimum; rect.offsetMax = maximum;
    }

    private static void Wire(Object target, string field, Object value)
    {
        var data = new SerializedObject(target); data.FindProperty(field).objectReferenceValue = value;
        data.ApplyModifiedPropertiesWithoutUndo();
    }

    private static void Tests(Type type)
    {
        int passed = 0, failed = 0, skipped = 0;
        foreach (var method in type.GetMethods())
        {
            var cases = method.GetCustomAttributes(typeof(TestCaseAttribute), false).Cast<TestCaseAttribute>().ToArray();
            if (cases.Length == 0 && method.GetCustomAttributes(typeof(TestAttribute), false).Length == 0) continue;
            var arguments = cases.Length > 0 ? cases.Select(c => c.Arguments) : new[] { Array.Empty<object>() };
            foreach (object[] args in arguments)
            {
                object fixture = Activator.CreateInstance(type);
                string name = type.Name + "." + method.Name + "(" + string.Join(",", args) + ")";
                try { type.GetMethod("SetUp")?.Invoke(fixture, null); method.Invoke(fixture, args); passed++; Log("PASS " + name); }
                catch (Exception error)
                {
                    Exception cause = error.InnerException ?? error;
                    if (cause is IgnoreException) { skipped++; Log("SKIP " + name); }
                    else { failed++; Log("FAIL " + name + " " + cause); }
                }
                finally { type.GetMethod("TearDown")?.Invoke(fixture, null); }
            }
        }
        Log("TESTS " + type.Name + " passed=" + passed + " failed=" + failed + " skipped=" + skipped);
    }
