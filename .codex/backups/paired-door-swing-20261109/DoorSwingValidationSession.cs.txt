using System;
using System.IO;
using System.Linq;
using System.Reflection;
using Photon.Pun;
using UnityEngine.AI;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.SceneManagement;

[InitializeOnLoad]
internal static class DoorSwingValidationSession
{
    private const string Request = "Temp/DoorSwing.request";
    private const string Result = ".codex/diagnostics/paired-door-swing-20261109.txt";
    private const string Backup = ".codex/backups/paired-door-swing-20261109/";
    private const string Preview = "Assets/Tests/DoorSwingValidation.unity";
    private const BindingFlags Private = BindingFlags.Instance | BindingFlags.NonPublic;
    private static DoorController door, other;
    private static Transform left, right, leftPivot, rightPivot;
    private static Quaternion leftRest, rightRest;
    private static Vector3 leftPosition, rightPosition, leftAnchor, rightAnchor;
    private static PlayerPickupInteractor interactor;
    private static int stage, opened, closed, squeaking;
    private static double began;
    private static Vector3 leftCenter, rightCenter, doorwayNormal, doorwayCenter;
    private static float desiredSide;
    static DoorSwingValidationSession()
    {
        EditorApplication.update += Update;
        EditorApplication.playModeStateChanged += phase =>
        {
            if (phase == PlayModeStateChange.EnteredPlayMode && SessionState.GetBool("DoorSwing.Play", false))
            {
                foreach (var manager in UnityEngine.Object.FindObjectsByType<RoomManager>(FindObjectsInactive.Include)) { manager.StopAllCoroutines(); manager.enabled = false; manager.gameObject.SetActive(false); }
                if (PhotonNetwork.IsConnected) PhotonNetwork.Disconnect();
            }
            if (phase != PlayModeStateChange.EnteredEditMode || !SessionState.GetBool("DoorSwing.Restore", false)) return;
            SessionState.SetBool("DoorSwing.Restore", false);
            var original = SceneManager.GetSceneByPath(SessionState.GetString("DoorSwing.Original", "Assets/Scenes/Map3.unity"));
            if (original.IsValid()) SceneManager.SetActiveScene(original);
            if (original.IsValid() && SessionState.GetBool("DoorSwing.Dirty", false)) EditorSceneManager.MarkSceneDirty(original);
            var preview = SceneManager.GetSceneByPath(Preview); if (preview.IsValid()) EditorSceneManager.CloseScene(preview, true);
            AssetDatabase.DeleteAsset(Preview); Log("RESTORED " + original.path + " dirty=" + original.isDirty);
        };
    }
    private static void Update()
    {
        if (SessionState.GetBool("DoorSwing.Play", false) && EditorApplication.isPlaying)
        {
            try { Tick(); } catch (Exception error) { Log("PLAY FAILED " + error); End(); }
            return;
        }
        if (!File.Exists(Request) || EditorApplication.isCompiling || EditorApplication.isUpdating || EditorApplication.isPlayingOrWillChangePlaymode) return;
        string command = File.ReadAllText(Request).Trim(); File.Delete(Request);
        try { if (command == "tests") Tests(); else if (command == "play") Begin(); }
        catch (Exception error) { Log("FAILED " + error); }
    }
    private static void Log(string value) => File.AppendAllText(Result, value + "\n");
    private static string PathOf(Transform value) => value.parent == null ? value.name : PathOf(value.parent) + "/" + value.name;
    private static T[] Find<T>(Scene scene) where T : Component => scene.GetRootGameObjects().SelectMany(root => root.GetComponentsInChildren<T>(true)).ToArray();
    private static Transform Arch(Scene scene) => Find<Transform>(scene).Single(t => t.name == "Door_Arch_B" && t.Find("Door_L_Arch_B") != null);
    private static void Set(object target, string name, object value) => target.GetType().GetField(name, Private).SetValue(target, value);
    private static void Tests()
    {
        var original = SceneManager.GetActiveScene(); bool dirty = original.isDirty;
        var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Additive); int passed = 0, failed = 0;
        try
        {
            foreach (var type in new[] { typeof(DoubleLeafDoorTests), typeof(EnemyTraversalTests), typeof(PlatformControllerTests), typeof(InteractionTriggerRoutingTests), typeof(InteractionTriggerSystemTests), typeof(TimedInteractionTests), typeof(PlayerPickupInteractorFocusTests), typeof(StatuePuzzleTests), typeof(FlamethrowerTrapTests) })
                foreach (var method in type.GetMethods().Where(m => m.GetCustomAttributes(typeof(NUnit.Framework.TestAttribute), false).Length > 0))
                {
                    if (type == typeof(EnemyTraversalTests) && method.Name != "Route_KeepsTheChosenDirection" && method.Name != "DoorProbe_IgnoresTheLeafBehindTheEnemyButFindsItAhead") continue;
                    var fixture = Activator.CreateInstance(type);
                    try { type.GetMethod("SetUp")?.Invoke(fixture, null); method.Invoke(fixture, null); passed++; }
                    catch (Exception error) { failed++; Log("TEST FAILED " + type.Name + "." + method.Name + " " + error); }
                    finally { type.GetMethod("TearDown")?.Invoke(fixture, null); }
                }
        }
        finally { SceneManager.SetActiveScene(original); EditorSceneManager.CloseScene(scene, true); if (dirty) EditorSceneManager.MarkSceneDirty(original); }
        Log("TESTS passed=" + passed + " failed=" + failed + " originalDirty=" + original.isDirty);
    }
    private static void Begin()
    {
        var original = SceneManager.GetActiveScene(); var arch = Arch(original);
        SessionState.SetString("DoorSwing.Original", original.path); SessionState.SetBool("DoorSwing.Dirty", original.isDirty);
        var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Additive);
        var clone = UnityEngine.Object.Instantiate(arch.gameObject); clone.name = arch.name; foreach (var member in clone.GetComponentsInChildren<DoorController>(true)) Set(member, "prototypeLocalOnly", true);
        var probe = new GameObject("Interaction Probe", typeof(Camera), typeof(PlayerPickupInteractor)); probe.GetComponent<Camera>().enabled = false;
        Set(probe.GetComponent<PlayerPickupInteractor>(), "interactionCamera", probe.GetComponent<Camera>());
        Set(probe.GetComponent<PlayerPickupInteractor>(), "showInteractionPrompt", false); Set(probe.GetComponent<PlayerPickupInteractor>(), "highlightPickupUnderAim", false);
        var light = new GameObject("Preview Light", typeof(Light)); light.GetComponent<Light>().type = LightType.Directional; light.transform.rotation = Quaternion.Euler(50, -30, 0);
        EditorSceneManager.SaveScene(scene, Preview); SessionState.SetBool("DoorSwing.Play", true); EditorApplication.isPlaying = true;
    }
    private static void Check(bool condition, string text) { if (!condition) throw new InvalidOperationException(text); Log("PASS " + text); }
    private static void Next() { stage++; began = Time.timeAsDouble; }
    private static float Angle(Transform part, Quaternion rest) => Quaternion.Angle(rest, part.localRotation);
    private static void Anchors()
    {
        Check(Vector3.Distance(leftPivot.position, leftAnchor) < 0.002f && Vector3.Distance(rightPivot.position, rightAnchor) < 0.002f, "Both authored hinge positions remain fixed");
    }
    private static Vector3 Center(Transform part)
    {
        var filter = part.GetComponentInChildren<MeshFilter>(true);
        return filter.transform.TransformPoint(filter.sharedMesh.bounds.center);
    }
    private static void SetDesiredSide() => desiredSide = -Mathf.Sign(Vector3.Dot(interactor.transform.position - doorwayCenter, doorwayNormal));
    private static void SamePhysicalSide()
    {
        float a = Vector3.Dot(Center(left) - leftCenter, doorwayNormal);
        float b = Vector3.Dot(Center(right) - rightCenter, doorwayNormal);
        Check(a * desiredSide > 0.001f && b * desiredSide > 0.001f, "Both panel centers move to the same side away from the actor; left=" + a + " right=" + b + " expected=" + desiredSide);
    }
    private static void Aim(Transform leaf, int sign)
    {
        var collider = leaf.GetComponent<BoxCollider>(); var local = collider.center;
        local.y = Mathf.Clamp(leaf.InverseTransformPoint(leaf.Find("Prompt_Pivot").position).y, collider.center.y - collider.size.y * 0.5f + 0.1f, collider.center.y + collider.size.y * 0.5f - 0.1f);
        var target = leaf.TransformPoint(local); var normal = leaf.TransformVector(Vector3.forward).normalized;
        foreach (int direction in new[] { sign, -sign })
        {
            var origin = target + normal * direction * 2;
            if (!Physics.Raycast(origin, (target - origin).normalized, out var hit, 2.5f) || hit.collider.transform != leaf) continue;
            interactor.transform.SetPositionAndRotation(origin, Quaternion.LookRotation(target - origin));
            var focus = typeof(PlayerPickupInteractor).GetMethod("ResolveInteractionFocus", Private).Invoke(interactor, null);
            Check(focus.GetType().GetProperty("Interactable").GetValue(focus) == (object)leaf.GetComponent<DoorController>(), "Aiming at " + leaf.name + " resolves its own root Door Controller"); return;
        }
        throw new InvalidOperationException("Cannot aim at " + leaf.name);
    }
    private static void Tick()
    {
        if (door == null)
        {
            var preview = SceneManager.GetSceneByPath(Preview);
            for (int i = 0; i < SceneManager.sceneCount; i++) if (SceneManager.GetSceneAt(i) != preview)
                foreach (var root in SceneManager.GetSceneAt(i).GetRootGameObjects()) root.SetActive(false);
            SceneManager.SetActiveScene(preview); var arch = Arch(preview);
            left = arch.Find("Door_L_Arch_B"); right = arch.Find("Door_R_Arch_B"); leftPivot = left.Find("Pivot"); rightPivot = right.Find("Pivot");
            door = left.GetComponent<DoorController>(); other = right.GetComponent<DoorController>();
            Check(arch.GetComponent<DoorController>() == null && door != other && door.PairedDoor == other && other.PairedDoor == door, "Both roots own equivalent reciprocal controllers");
            leftRest = left.localRotation; rightRest = right.localRotation; leftPosition = left.localPosition; rightPosition = right.localPosition; leftAnchor = leftPivot.position; rightAnchor = rightPivot.position;
            leftCenter = Center(left); rightCenter = Center(right); doorwayNormal = arch.forward; doorwayCenter = (leftCenter + rightCenter) * 0.5f;
            interactor = Find<PlayerPickupInteractor>(preview).Single(); typeof(PhotonView).GetProperty("IsMine").SetValue(interactor.GetComponent<PhotonView>(), true);
            Set(interactor, "interactionCamera", interactor.GetComponent<Camera>());
            Check(door.TryGetInteractionPromptPlacement(default, out var leftPrompt) && leftPrompt.Anchor == left.Find("Prompt_Pivot")
                && other.TryGetInteractionPromptPlacement(default, out var rightPrompt) && rightPrompt.Anchor == right.Find("Prompt_Pivot"), "Each controller retains its own authored prompt anchor");
            Check(arch.GetComponent<ReactionSignalReceiver>() == null && arch.GetComponent<ReactionSignalEmitter>() == null && arch.GetComponent<DoorReactionSignalBridge>() == null, "The static arch has no door reaction stack");
            foreach (var receiver in arch.GetComponentsInChildren<ReactionSignalReceiver>())
            {
                var entries = (System.Collections.Generic.List<ReactionSignalReceiver.ReactionSignalEntry>)typeof(ReactionSignalReceiver).GetField("signalEntries", Private).GetValue(receiver);
                foreach (var entry in entries) entry.audioCue = null;
            }
            opened = closed = squeaking = 0;
            foreach (var member in new[] { door, other }) { member.Opened += _ => opened++; member.Closed += _ => closed++; member.Squeaking += _ => squeaking++; }
            Physics.SyncTransforms(); Aim(left, 1); SetDesiredSide(); Check(interactor.TryInteract(), "One player interaction begins opening both leaves");
            Check(!interactor.TryInteract() && door.IsOpen, "Shared cooldown prevents a second toggle"); stage = 0; began = Time.timeAsDouble;
        }
        if (Time.timeAsDouble - began > 20) throw new TimeoutException("Door stage " + stage);
        switch (stage)
        {
            case 0:
                if (Angle(left, leftRest) < 3) return;
                Check(Mathf.Abs(Angle(left, leftRest) - Angle(right, rightRest)) < 0.05f && Angle(right, rightRest) < 89, "Both leaves move together during partial opening"); Anchors(); SamePhysicalSide(); Next(); break;
            case 1:
                if (door.IsMoving) return;
                Check(Angle(left, leftRest) > 89.9f && Angle(right, rightRest) > 89.9f && opened == 2, "One open command reaches 90 degrees on both leaves"); Anchors(); SamePhysicalSide(); Capture("open");
                Aim(right, 1); Check(interactor.TryInteract() && !door.IsOpen, "Interaction on the right leaf closes the whole door"); Next(); break;
            case 2:
                if (door.IsMoving) return;
                Check(Vector3.Distance(left.localPosition, leftPosition) < 0.002f && Vector3.Distance(right.localPosition, rightPosition) < 0.002f && closed == 2, "Both leaves return exactly to their authored closed poses"); Anchors(); Capture("closed");
                Aim(right, -1); SetDesiredSide(); Check(interactor.TryInteract() && door.IsOpen, "The right leaf also opens both leaves from the opposite side"); Next(); break;
            case 3:
                float duration = (float)typeof(DoorController).GetField("moveDuration", Private).GetValue(door);
                if (Time.timeAsDouble - began < Math.Max(1.05, duration * 0.75)) return;
                Check(door.IsMoving && Mathf.Abs(Angle(left, leftRest) - Angle(right, rightRest)) < 0.05f, "Both leaves remain synchronized from the opposite side"); SamePhysicalSide();
                Aim(left, 1); Check(interactor.TryInteract() && !door.IsOpen, "A player interaction can reverse both leaves during travel"); Next(); break;
            case 4:
                if (door.IsMoving) return;
                Anchors(); Check(Angle(left, leftRest) < 0.01f && Angle(right, rightRest) < 0.01f && opened == 4 && closed == 4 && squeaking == 8, "Each peer emits its own reactions once per transition");
                Log("PLAY PASSED"); End(); break;
        }
    }
    private static void Capture(string suffix)
    {
        var go = new GameObject("Door Preview Camera", typeof(Camera)); var camera = go.GetComponent<Camera>();
        go.transform.position = door.transform.parent.position + new Vector3(0, 4.5f, 11); go.transform.LookAt(door.transform.parent.position + Vector3.up * 4.5f);
        camera.orthographic = true; camera.orthographicSize = 5.3f; camera.nearClipPlane = 0.01f; camera.farClipPlane = 50;
        var texture = RenderTexture.GetTemporary(1000, 1000, 24); var previous = RenderTexture.active;
        try
        {
            camera.targetTexture = texture; camera.Render(); RenderTexture.active = texture;
            var image = new Texture2D(1000, 1000, TextureFormat.RGB24, false); image.ReadPixels(new Rect(0, 0, 1000, 1000), 0, 0); image.Apply();
            File.WriteAllBytes(".codex/diagnostics/door-swing-" + suffix + ".png", image.EncodeToPNG()); UnityEngine.Object.DestroyImmediate(image);
        }
        finally { camera.targetTexture = null; RenderTexture.active = previous; RenderTexture.ReleaseTemporary(texture); UnityEngine.Object.DestroyImmediate(go); }
    }
    private static void End() { SessionState.SetBool("DoorSwing.Play", false); SessionState.SetBool("DoorSwing.Restore", true); EditorApplication.isPlaying = false; }
    private static void Inspect(bool snapshot)
    {
        var scene = SceneManager.GetActiveScene(); bool dirty = scene.isDirty;
        if (snapshot)
        {
            SessionState.SetBool("DoorSwing.AuthorDirty", dirty);
            EditorSceneManager.SaveScene(scene, Backup + "Map3.live-before.unity", true);
            if (dirty) EditorSceneManager.MarkSceneDirty(scene);
        }
        Log("SCENE " + scene.path + " dirty=" + dirty);
        var transforms = scene.GetRootGameObjects().SelectMany(root => root.GetComponentsInChildren<Transform>(true)).ToArray();
        foreach (var door in transforms.Select(t => t.GetComponent<DoorController>()).Where(d => snapshot && d != null))
        {
            Log("EXISTING " + PathOf(door.transform) + " " + EditorJsonUtility.ToJson(door));
            Log("STACK " + string.Join(",", door.GetComponents<Component>().Select(c => c == null ? "Missing" : c.GetType().Name)));
        }
        foreach (var candidate in transforms.Where(t => t.name == "Door_Arch_B")) Log("CANDIDATE " + PathOf(candidate) + " descendants=" + candidate.GetComponentsInChildren<Transform>(true).Length);
        var arch = transforms.Single(t => t.name == "Door_Arch_B" && t.GetComponentsInChildren<Transform>(true).Any(c => c.name == "Door_L_Arch_B"));
        foreach (var item in arch.GetComponentsInChildren<Transform>(true))
        {
            Log("OBJECT " + PathOf(item) + " position=" + item.position.ToString("F5") + " localPosition=" + item.localPosition.ToString("F5") + " localRotation=" + item.localEulerAngles.ToString("F5") + " scale=" + item.lossyScale.ToString("F5"));
            foreach (var component in item.GetComponents<Component>())
            {
                if (component is Renderer renderer) Log("RENDERER " + renderer.bounds);
                else if (component is not Transform) Log("COMPONENT " + component.GetType().Name + " " + EditorJsonUtility.ToJson(component));
            }
        }
    }
}
